cmake_minimum_required(VERSION 3.20)
project(MaaRocoAgent LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MAA_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../deps" CACHE PATH "MaaFramework SDK")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR (NOT WIN32 AND NOT APPLE))
  message(FATAL_ERROR "MaaRoco supports Windows x64 and macOS arm64/x64")
endif()
foreach(lib MaaAgentServer MaaFramework)
  add_library(${lib} SHARED IMPORTED)
  if(WIN32)
    set_target_properties(${lib} PROPERTIES IMPORTED_IMPLIB "${MAA_SDK}/lib/${lib}.lib"
      IMPORTED_LOCATION "${MAA_SDK}/bin/${lib}.dll")
  else()
    set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${MAA_SDK}/bin/lib${lib}.dylib")
  endif()
endforeach()
if(WIN32)
  set(ROCO_INPUT cpp/input.cpp)
  set(ROCO_LAUNCH cpp/launch.cpp)
  set(ROCO_RUNTIME win-x64)
else()
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  set(ROCO_INPUT cpp/input_macos.mm)
  set(ROCO_LAUNCH cpp/launch_macos.cpp)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(ROCO_RUNTIME osx-arm64)
  else()
    set(ROCO_RUNTIME osx-x64)
  endif()
endif()
add_library(roco_common STATIC cpp/common.cpp ${ROCO_INPUT})
target_include_directories(roco_common PUBLIC "${MAA_SDK}/include" cpp cpp/third_party/meojson/include)
if(WIN32)
  target_compile_definitions(roco_common PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_link_libraries(roco_common PUBLIC user32)
else()
  target_link_libraries(roco_common PUBLIC "-framework AppKit" "-framework ApplicationServices")
endif()
add_executable(MaaRocoAgent cpp/main.cpp ${ROCO_LAUNCH} cpp/target_pet.cpp)
target_link_libraries(MaaRocoAgent PRIVATE roco_common MaaAgentServer)
set(ROCO_INSTALL_TARGETS MaaRocoAgent)
if(WIN32)
  add_executable(MaaRocoRunner cpp/runner.cpp)
  target_link_libraries(MaaRocoRunner PRIVATE roco_common MaaFramework)
  list(APPEND ROCO_INSTALL_TARGETS MaaRocoRunner)
else()
  set_target_properties(MaaRocoAgent PROPERTIES BUILD_RPATH "${MAA_SDK}/bin" INSTALL_RPATH "@loader_path")
  add_executable(MaaRocoLauncher cpp/launcher_macos.cpp)
  install(TARGETS MaaRocoLauncher RUNTIME DESTINATION ../MacOS)
endif()
add_executable(MaaRocoAgentTests cpp/tests.cpp)
target_link_libraries(MaaRocoAgentTests PRIVATE roco_common)
foreach(target roco_common ${ROCO_INSTALL_TARGETS} MaaRocoAgentTests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /O1)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Os)
    if(WIN32)
      target_link_options(${target} PRIVATE -static -s)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME agent_strategy COMMAND MaaRocoAgentTests)
install(TARGETS ${ROCO_INSTALL_TARGETS} RUNTIME DESTINATION runtimes/${ROCO_RUNTIME}/native)
